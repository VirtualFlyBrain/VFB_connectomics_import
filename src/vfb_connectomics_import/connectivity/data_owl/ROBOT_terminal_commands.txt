# n2n builds no longer use ROBOT. Each connectivity CLI writes the OWL directly and
# validates it (--output connectome_X.owl; see docs/CONNECTIVITY_BUILDS.md), e.g.
#   python -m vfb_connectomics_import.connectivity.cli.banc --site BANC888 \
#       --output connectome_BANC_n2n.owl --previous data_owl/connectome_BANC_n2n.owl.gz
# CI checks that output against `robot template` (robot diff: identical ontologies).
#
# The commands below are the historical ROBOT builds, kept for reference. Output
# must be connectome_*.owl to be side-loaded. A CLI still writes a ROBOT template
# for --output X.tsv (the release pipeline's CATMAID step); stamp its release by
# hand with the values in X.tsv.provenance:
#   annotate --ontology-iri http://virtualflybrain.org/data/VFB/OWL/<name>.owl \
#            --version-iri <VERSION_IRI> --annotation dc:source "<DC_SOURCE>"

robot template -input-iri http://purl.obolibrary.org/obo/ro.owl --add-prefix "n2o: http://neo2owl/custom/" --add-prefix "VFB: http://virtualflybrain.org/reports/VFB_" --template /Users/alexmclachlan/Documents/GitHub/VFB_connectomics_import/Robot_template.tsv  annotate --ontology-iri http://virtualflybrain.org/data/VFB/OWL/CATMAID_import.owl convert -f ofn --output MANC_1_2_1_n2n_connectivity_Import.owl

robot template -input-iri http://purl.obolibrary.org/obo/ro.owl --add-prefix "n2o: http://neo2owl/custom/" --add-prefix "VFB: http://virtualflybrain.org/reports/VFB_" --template /Users/adm/Documents/GitHub/VFB_connectomics_import/Nern2024_n_2_n.tsv annotate --ontology-iri http://virtualflybrain.org/data/VFB/OWL/CATMAID_import.owl convert -f ofn --output connectome_MCNSOL_1_0_n2n.owl

robot template -input-iri http://purl.obolibrary.org/obo/ro.owl --add-prefix "n2o: http://neo2owl/custom/" --add-prefix "VFB: http://virtualflybrain.org/reports/VFB_" --template /Users/adm/Documents/GitHub/VFB_connectomics_import/Berg2025_n_2_n.tsv annotate --ontology-iri http://virtualflybrain.org/data/VFB/OWL/CATMAID_import.owl convert -f ofn --output connectome_malecns_1_0_n2n.owl
